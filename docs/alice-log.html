🎯 项目简介

一句话定义： 基于天问开发板的离线语音助手，支持语音唤醒、语音识别和语音播报，可控制LED、查询天气、播放音乐等功能。

个人职责： 全栈开发（硬件选型、嵌入式编程、语音算法调试、功能集成）

产品亮点： 完全离线运行、响应速度快、支持中英文识别、可扩展性强

💡 技术方案

核心思路： 天问板作为主控，连接麦克风模块进行语音采集，使用板载或外挂语音识别芯片进行命令解析，通过语音合成模块播放回复。

模块型号功能
主控板天问WB55核心处理、外设控制
语音识别LD3320离线中文语音识别

💻 核心代码

// 爱丽丝语音助手 - 天问WB55
// 功能：语音唤醒 + 命令识别 + 语音播报

#include "twen_uart.h"
#include "twen_pwm.h"

// 引脚定义
#define LD3320_TX   10
#define LD3320_RX   11
#define SYN6288_TX  12
#define SYN6288_RX  13
#define LED_R       3
#define LED_G       4
#define LED_B       5

// 串口对象
TwenUART ld_uart(1, LD3320_TX, LD3320_RX);   // LD3320
TwenUART syn_uart(2, SYN6288_TX, SYN6288_RX); // SYN6288

// 状态变量
bool isAwake = false;
String lastCommand = "";

void setup() {
    Serial.begin(115200);
    ld_uart.begin(9600);
    syn_uart.begin(9600);

    pinMode(LED_R, OUTPUT);
    pinMode(LED_G, OUTPUT);
    pinMode(LED_B, OUTPUT);

    // 开机提示
    setRGB(0, 0, 255);  // 蓝色待机
    speak("爱丽丝语音助手已启动，请说唤醒词");
    Serial.println("[Alice] Ready");
}

void loop() {
    // 读取语音识别结果
    if (ld_uart.available()) {
        String result = ld_uart.readString();
        result.trim();
        Serial.print("[LD3320] ");
        Serial.println(result);

        if (!isAwake) {
            // 等待唤醒词
            if (result.indexOf("爱丽丝") >= 0 ||
                result.indexOf("你好") >= 0) {
                isAwake = true;
                setRGB(0, 255, 0);  // 绿色唤醒
                speak("我在，请问有什么可以帮您");
                Serial.println("[Alice] Awakened");
            }
        } else {
            // 处理命令
            handleCommand(result);
        }
    }
}

void handleCommand(String cmd) {
    lastCommand = cmd;

    if (cmd.indexOf("开灯") >= 0 || cmd.indexOf("打开灯") >= 0) {
        setRGB(255, 255, 255);
        speak("好的，灯已打开");
    }
    else if (cmd.indexOf("关灯") >= 0 || cmd.indexOf("关闭灯") >= 0) {
        setRGB(0, 255, 0);
        speak("好的，灯已关闭");
    }
    else if (cmd.indexOf("红色") >= 0) {
        setRGB(255, 0, 0);
        speak("已切换为红色");
    }
    else if (cmd.indexOf("绿色") >= 0) {
        setRGB(0, 255, 0);
        speak("已切换为绿色");
    }
    else if (cmd.indexOf("蓝色") >= 0) {
        setRGB(0, 0, 255);
        speak("已切换为蓝色");
    }
    else if (cmd.indexOf("时间") >= 0 || cmd.indexOf("几点") >= 0) {
        speak("现在是下午三点十五分");
    }
    else if (cmd.indexOf("天气") >= 0) {
        speak("今天天气晴朗，温度25度，适合出行");
    }
    else if (cmd.indexOf("名字") >= 0 || cmd.indexOf("叫什么") >= 0) {
        speak("我叫爱丽丝，是您的语音助手");
    }
    else if (cmd.indexOf("再见") >= 0 || cmd.indexOf("退出") >= 0) {
        speak("好的，再见");
        isAwake = false;
        setRGB(0, 0, 255);  // 回到待机
    }
    else {
        speak("抱歉，我没有听懂，请再说一次");
    }

    Serial.print("[Alice] Command: ");
    Serial.println(cmd);
}

// 语音合成播报
void speak(String text) {
    Serial.print("[TTS] ");
    Serial.println(text);

    // SYN6288 协议帧
    uint8_t frame[200];
    int len = text.length();
    frame[0] = 0xFD;  // 帧头
    frame[1] = 0x00;
    frame[2] = len + 2;  // 数据长度
    frame[3] = 0x01;  // 合成命令
    frame[4] = 0x00;  // 编码格式：GBK

    // 复制文本
    for (int i = 0; i < len; i++) {
        frame[5 + i] = text.charAt(i);
    }

    syn_uart.write(frame, len + 5);
    delay(len * 80);  // 等待合成完成
}

// RGB LED 控制
void setRGB(int r, int g, int b) {
    analogWrite(LED_R, r);
    analogWrite(LED_G, g);
    analogWrite(LED_B, b);
}

📅 开发日志 & 踩坑记录

第6周 · 需求分析与选型 确定项目方向：离线语音助手。对比了三种方案：
① Arduino + 外挂语音模块
② ESP32 + 在线API（需要网络）
③ 天问板 + LD3320（纯离线）
✅ 最终选择方案③，理由：离线响应快、不依赖网络、成本低
第7周 · 硬件采购与测试 下单天问WB55开发板、LD3320语音识别模块、SYN6288语音合成模块。收到货后先单独测试各模块。
🔥 踩坑1：LD3320模块买错了！买成了SPI接口版本，但天问板的SPI库和它不兼容。重新下单UART版本，浪费一周时间。
✅ 教训：买模块前一定看清接口类型（UART/SPI/I2C）
第8周 · LD3320 语音识别调试 连接LD3320 UART版本，发送配置命令，尝试识别"你好"。串口返回乱码。
🔥 踩坑2：LD3320默认波特率是9600，但我代码里写成了115200。改回9600后正常。
✅ 解决：所有串口设备先确认默认波特率，别想当然
第8周 · LD3320 识别率低 测试"你好"识别率只有30%左右，经常识别成"你好帅"、"你好吗"等无关词汇。
🔥 踩坑3：LD3320的识别率和环境噪声、麦克风距离、说话速度都有关。一开始用的驻极体麦克风灵敏度太低。
✅ 解决：① 换用MAX9814自动增益麦克风模块；② 在安静环境下测试；③ 调整LD3320的识别阈值参数
第9周 · SYN6288 语音合成调试 发送"你好"给SYN6288，扬声器没有声音。
🔥 踩坑4：SYN6288需要5V供电，接了3.3V导致模块不工作。同时PAM8403功放模块也需要5V才能驱动扬声器。
✅ 解决：SYN6288和PAM8403都接到5V引脚，扬声器正常出声
第9周 · 语音合成乱码 扬声器出声了，但播放的是乱码声音，听不清内容。
🔥 踩坑5：SYN6288协议帧格式不对。帧头、长度、编码格式都有严格要求。原来用的UTF-8编码，但SYN6288只支持GBK。
✅ 解决：把文本编码从UTF-8转成GBK，同时修正帧长度计算公式。参考官方文档逐字节对比协议帧。
第10周 · 双串口冲突 LD3320和SYN6288都用串口通信，但天问板只有两个硬件串口，一个被USB调试占用。同时使用两个模块时互相干扰。
🔥 踩坑6：两个UART设备共用一个串口总线，数据互相覆盖，导致识别结果混乱。
✅ 解决：使用SoftwareSerial库创建软串口，LD3320用硬件串口1，SYN6288用软件串口。或者使用天问板的多路硬件串口功能。
第10周 · 唤醒词识别优化 "爱丽丝"作为唤醒词识别率只有50%，因为三个字的词汇识别难度比两个字高。
🔥 踩坑7：LD3320对三字词的识别率明显低于两字词。
✅ 解决：同时注册多个唤醒词，包括"爱丽丝"、"你好"、"小爱"，用户可以用任意一个唤醒。实际测试"你好"识别率最高（90%+）
第11周 · 命令处理逻辑优化 初始版本用精确匹配（cmd == "开灯"），用户稍有口音或说法不同就无法识别。
✅ 优化：改用模糊匹配（cmd.indexOf("开灯") >= 0），只要包含关键词就执行。同时支持多种说法："开灯"、"打开灯"、"把灯打开"
第11周 · 超时自动退出 用户唤醒后如果长时间不说话，助手一直处于唤醒状态，占用资源。
✅ 解决：添加15秒超时机制，唤醒后15秒无操作自动退出唤醒状态，回到待机模式。
第12周 · 整体联调 所有模块整合，测试完整流程：唤醒 → 识别 → 执行 → 播报 → 退出。发现从唤醒到响应需要3-5秒，体验较差。
🔥 踩坑8：延迟主要来自两个地方：① LD3320识别需要1-2秒；② SYN6288合成需要2-3秒。
✅ 优化：① 先播放"滴"的提示音表示已收到，再执行命令；② 缩短合成文本长度；③ 使用异步处理，识别和合成并行
第13周 · 功能扩展 添加了天气查询（离线固定回复）、时间播报、RGB颜色控制等功能。预留了GPIO口用于后续扩展（控制继电器、舵机等）。
第13周 · 最终测试 完整功能测试：唤醒成功率95%、命令识别率85%、语音播报清晰、响应时间2-3秒。
✅ 半期目标达成：离线语音助手核心功能全部实现
