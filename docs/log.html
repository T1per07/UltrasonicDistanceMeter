💻 核心代码 (Arduino IDE)

// ESP32-C3 超声波测距 + 单 LED 提示
#define TRIG_PIN  5
#define ECHO_PIN  18
#define LED_PIN   19

void setup() {
  Serial.begin(115200);
  pinMode(TRIG_PIN, OUTPUT);
  pinMode(ECHO_PIN, INPUT);
  pinMode(LED_PIN, OUTPUT);
  Serial.println("Ultrasonic ready");
}

void loop() {
  digitalWrite(TRIG_PIN, LOW);
  delayMicroseconds(2);
  digitalWrite(TRIG_PIN, HIGH);
  delayMicroseconds(10);
  digitalWrite(TRIG_PIN, LOW);

  long duration = pulseIn(ECHO_PIN, HIGH);
  float distance = duration * 0.034 / 2;

  Serial.print("Distance: ");
  Serial.print(distance);
  Serial.println(" cm");

  if (distance > 0 && distance < 30) {
    digitalWrite(LED_PIN, HIGH);   // 点亮LED
  } else {
    digitalWrite(LED_PIN, LOW);    // 熄灭LED
  }
  delay(200);
}

代码说明： 每200ms测量一次距离，通过串口输出实时值，同时根据阈值（30cm）控制LED亮灭。项目不需要蜂鸣器，安静且聚焦核心功能。

🔥 踩坑2：ESP32-C3 需要手动进入下载模式 —— 按住 BOOT 键，点上传，等出现 "Connecting..." 后松开 BOOT。之前不知道这个操作，白耗了20分钟。
✅ 解决：按住 BOOT → 点上传 → 松开 BOOT，成功率100%
周六 11:40 · 串口监视器显示乱码 上传成功后打开串口监视器，全是乱码。
🔥 踩坑3：串口波特率设置错误。代码写的是 115200，但串口监视器默认是 9600。改为 115200 后正常显示。
周六 13:30 · 首次硬件搭建 符永儒开始接线。按照查好的引脚图，Trig→GPIO4, Echo→GPIO5, LED→GPIO6。杜邦线直连，不用面包板。
🔥 踩坑4：HC-SR04 的 Echo 引脚输出 5V 高电平，但 ESP32-C3 的 GPIO 只耐受 3.3V。测试时发现偶尔会读到异常大的数值（>1000cm）。
✅ 解决：在 Echo 和 GPIO5 之间加了一个简单的分压电路（1kΩ + 2kΩ），把 5V 降到 3.3V。后来发现其实直接接也能跑，但为了板子安全还是保留了分压。
周六 14:20 · 测距数据跳变严重 串口输出的距离值每隔几秒就会跳变一次，比如正常是 25cm，突然蹦出一个 400cm 或 0cm。
🔥 踩坑5：超声波在光滑表面（如玻璃、白墙）反射不稳定，偶尔收不到回波。pulseIn() 超时返回 0，导致计算出距离 0。
✅ 解决：加了一个简单的滤波 —— 连续读5次取中位数，同时过滤掉 0 和 >400cm 的异常值。代码如下： 
float getStableDistance() {
  float readings[5];
  for (int i = 0; i < 5; i++) {
    digitalWrite(TRIG_PIN, LOW);
    delayMicroseconds(2);
    digitalWrite(TRIG_PIN, HIGH);
    delayMicroseconds(10);
    digitalWrite(TRIG_PIN, LOW);
    long dur = pulseIn(ECHO_PIN, HIGH, 30000);
    readings[i] = (dur > 0) ? dur * 0.034 / 2 : 400;
    delay(30);
  }
  // 冒泡排序取中位数
  for (int i = 0; i < 4; i++)
    for (int j = 0; j < 4-i; j++)
      if (readings[j] > readings[j+1]) {
        float t = readings[j];
        readings[j] = readings[j+1];
        readings[j+1] = t;
      }
  return readings[2];
}

// 爱丽丝语音助手 - 天问WB55
// 功能：语音唤醒 + 命令识别 + 语音播报

#include "twen_uart.h"
#include "twen_pwm.h"

// 引脚定义
#define LD3320_TX   10
#define LD3320_RX   11
#define SYN6288_TX  12
#define SYN6288_RX  13
#define LED_R       3
#define LED_G       4
#define LED_B       5

// 串口对象
TwenUART ld_uart(1, LD3320_TX, LD3320_RX);   // LD3320
TwenUART syn_uart(2, SYN6288_TX, SYN6288_RX); // SYN6288

// 状态变量
bool isAwake = false;
String lastCommand = "";

void setup() {
    Serial.begin(115200);
    ld_uart.begin(9600);
    syn_uart.begin(9600);

    pinMode(LED_R, OUTPUT);
    pinMode(LED_G, OUTPUT);
    pinMode(LED_B, OUTPUT);

    setRGB(0, 0, 255);  // 蓝色待机
    speak("爱丽丝语音助手已启动，请说唤醒词");
}

void loop() {
    if (ld_uart.available()) {
        String result = ld_uart.readString();
        result.trim();

        if (!isAwake) {
            if (result.indexOf("爱丽丝") >= 0 ||
                result.indexOf("你好") >= 0) {
                isAwake = true;
                setRGB(0, 255, 0);
                speak("我在，请问有什么可以帮您");
            }
        } else {
            handleCommand(result);
        }
    }
}

void handleCommand(String cmd) {
    if (cmd.indexOf("开灯") >= 0) {
        setRGB(255, 255, 255);
        speak("好的，灯已打开");
    }
    else if (cmd.indexOf("关灯") >= 0) {
        setRGB(0, 255, 0);
        speak("好的，灯已关闭");
    }
    else if (cmd.indexOf("红色") >= 0) {
        setRGB(255, 0, 0);
        speak("已切换为红色");
    }
    else if (cmd.indexOf("再见") >= 0) {
        speak("好的，再见");
        isAwake = false;
        setRGB(0, 0, 255);
    }
    else {
        speak("抱歉，我没有听懂");
    }
}

// 语音合成播报
void speak(String text) {
    uint8_t frame[200];
    int len = text.length();
    frame[0] = 0xFD;
    frame[1] = 0x00;
    frame[2] = len + 2;
    frame[3] = 0x01;
    frame[4] = 0x00;  // GBK编码
    for (int i = 0; i < len; i++) {
        frame[5 + i] = text.charAt(i);
    }
    syn_uart.write(frame, len + 5);
    delay(len * 80);
}

void setRGB(int r, int g, int b) {
    analogWrite(LED_R, r);
    analogWrite(LED_G, g);
    analogWrite(LED_B, b);
}

第6周 · 需求分析与选型 确定项目方向：离线语音助手。对比了三种方案：
① Arduino + 外挂语音模块
② ESP32 + 在线API（需要网络）
③ 天问板 + LD3320（纯离线）
✅ 最终选择方案③，理由：离线响应快、不依赖网络、成本低
第7周 · 硬件采购与测试 下单天问WB55开发板、LD3320语音识别模块、SYN6288语音合成模块。收到货后先单独测试各模块。
🔥 踩坑1：LD3320模块买错了！买成了SPI接口版本，但天问板的SPI库和它不兼容。重新下单UART版本，浪费一周时间。
✅ 教训：买模块前一定看清接口类型（UART/SPI/I2C）
第8周 · LD3320 语音识别调试 连接LD3320 UART版本，发送配置命令，尝试识别"你好"。串口返回乱码。
🔥 踩坑2：LD3320默认波特率是9600，但我代码里写成了115200。改回9600后正常。
✅ 解决：所有串口设备先确认默认波特率，别想当然
第8周 · LD3320 识别率低 测试"你好"识别率只有30%左右，经常识别成"你好帅"、"你好吗"等无关词汇。
🔥 踩坑3：LD3320的识别率和环境噪声、麦克风距离、说话速度都有关。一开始用的驻极体麦克风灵敏度太低。
✅ 解决：① 换用MAX9814自动增益麦克风模块；② 在安静环境下测试；③ 调整LD3320的识别阈值参数
第9周 · SYN6288 语音合成调试 发送"你好"给SYN6288，扬声器没有声音。
🔥 踩坑4：SYN6288需要5V供电，接了3.3V导致模块不工作。同时PAM8403功放模块也需要5V才能驱动扬声器。
✅ 解决：SYN6288和PAM8403都接到5V引脚，扬声器正常出声
第9周 · 语音合成乱码 扬声器出声了，但播放的是乱码声音，听不清内容。
🔥 踩坑5：SYN6288协议帧格式不对。帧头、长度、编码格式都有严格要求。原来用的UTF-8编码，但SYN6288只支持GBK。
✅ 解决：把文本编码从UTF-8转成GBK，同时修正帧长度计算公式。参考官方文档逐字节对比协议帧。
第10周 · 双串口冲突 LD3320和SYN6288都用串口通信，但天问板只有两个硬件串口，一个被USB调试占用。同时使用两个模块时互相干扰。
🔥 踩坑6：两个UART设备共用一个串口总线，数据互相覆盖，导致识别结果混乱。
✅ 解决：使用SoftwareSerial库创建软串口，LD3320用硬件串口1，SYN6288用软件串口。
第10周 · 唤醒词识别优化 "爱丽丝"作为唤醒词识别率只有50%，因为三个字的词汇识别难度比两个字高。
🔥 踩坑7：LD3320对三字词的识别率明显低于两字词。
✅ 解决：同时注册多个唤醒词，包括"爱丽丝"、"你好"、"小爱"，用户可以用任意一个唤醒。
第11周 · 命令处理逻辑优化 初始版本用精确匹配，用户稍有口音或说法不同就无法识别。
✅ 优化：改用模糊匹配，只要包含关键词就执行。同时支持多种说法。
第11周 · 超时自动退出 用户唤醒后如果长时间不说话，助手一直处于唤醒状态。
✅ 解决：添加15秒超时机制，唤醒后15秒无操作自动退出唤醒状态。
第12周 · 整体联调 所有模块整合，测试完整流程。发现从唤醒到响应需要3-5秒，体验较差。
🔥 踩坑8：延迟主要来自：① LD3320识别需要1-2秒；② SYN6288合成需要2-3秒。
✅ 优化：先播放"滴"的提示音表示已收到，再执行命令。
第13周 · 最终测试 完整功能测试：唤醒成功率95%、命令识别率85%、语音播报清晰、响应时间2-3秒。
✅ 半期目标达成：离线语音助手核心功能全部实现

📐 Week 1 · 计算机辅助设计（CAD）

本周目标： 学习使用 Fusion 360 进行三维建模，完成第一个零件设计。

主要内容：

  • Fusion 360 界面熟悉：草图、拉伸、旋转、倒角
  • 参数化设计思维：尺寸约束、几何约束
  • 工程图导出：三视图、剖视图标注
周三 · 软件安装与环境配置 安装 Fusion 360，注册教育版账号。第一次打开软件，界面比较复杂，花了半小时熟悉各个面板。
周四 · 草图绘制练习 学习草图工具：直线、圆、矩形、样条曲线。练习画一个简单的法兰盘。
🔥 踩坑：草图没有完全约束（变成绿色），拉伸时出现扭曲。解决：检查每个尺寸和约束是否标注完整。
周五 · 完成第一个零件 设计了一个简单的手机支架，使用拉伸、抽壳、倒角等命令。导出 STL 文件用于后续 3D 打印。
✅ 成果：成功导出 STL，零件尺寸符合预期

✂️ Week 2 · 激光切割

本周目标： 学习激光切割原理，设计并切割一个亚克力盒子。

主要内容：

  • 激光切割机工作原理：CO2激光、功率与速度的关系
  • 2D设计软件：AutoCAD / CorelDRAW
  • 榫卯结构设计：活动铰链、拼接卡扣
周三 · 激光切割机操作 学习安全规范（不能切割PVC！会释放氯气），熟悉对焦、调功率、调速度。用废料测试了切割和雕刻两种模式。
周四 · 设计盒子 用 AutoCAD 画了一个 80x60x40mm 的盒子，四面用榫卯拼接。设计时留了 0.1mm 的间隙补偿。
🔥 踩坑：第一版间隙留的太小（0.05mm），亚克力板塞不进去。改大到 0.15mm 后完美拼接。
周五 · 切割与组装 使用 3mm 亚克力板，功率 60%，速度 15mm/s。切割出来边缘光滑，拼接后结构稳固。
✅ 成果：激光切割亚克力盒子，可用于后续项目外壳

🖨️ Week 3 · 3D扫描与打印

本周目标： 学习 FDM 3D 打印原理，完成模型切片与打印。

主要内容：

  • FDM 打印原理：热熔堆积、层高、填充率
  • 切片软件：Bambu Studio / Cura
  • 支撑结构：何时需要支撑、支撑类型选择
  • 3D扫描：结构光扫描、点云处理
周三 · 打印机操作 学习 Bambu Lab 打印机的操作：装料、调平、开始打印。第一次看到打印机工作，很神奇。
周四 · 切片参数调试 测试不同参数：层高 0.1mm vs 0.2mm、填充率 20% vs 50%、有无支撑。发现 0.2mm 层高 + 20% 填充是最平衡的选择。
🔥 踩坑：打印一个悬空角度 60° 的模型，没加支撑，中间塌了。以后超过 45° 必须加支撑。
周五 · 完成打印作品 打印了 Week 1 设计的手机支架，PLA 材质，香芋紫色。打印时间约 2 小时，成品质量不错。
✅ 成果：第一个 3D 打印实物作品

🔌 Week 4 · 电子设计入门（PCB）

本周目标： 学习使用 KiCad 绘制原理图和 PCB 板。

主要内容：

  • 电路基础：欧姆定律、串并联、分压电路
  • KiCad 软件：原理图绘制、PCB布局、布线
  • 元器件封装：0805电阻、SOT-23三极管、ESP32模组
周三 · 电路基础回顾 复习欧姆定律（U=IR），计算 LED 限流电阻值。3.3V 供电，LED 压降 2V，目标电流 15mA → R = (3.3-2)/0.015 ≈ 87Ω，选用 100Ω。
周四 · KiCad 原理图 绘制 ESP32-C3 最小系统原理图：主控芯片、USB接口、LDO稳压、LED指示灯。
🔥 踩坑：找不到 ESP32-C3 的 KiCad 库。解决：从嘉立创下载官方封装库导入。
周五 · PCB 布局布线 完成 PCB 布局：晶振靠近芯片、USB 差分线等长、地平面完整。布线花了 3 小时。
✅ 成果：第一块自己设计的 PCB，导出 Gerber 文件

⚡ Week 5 · 电子制作（焊接）

本周目标： 学习手工焊接技术，制作 Arduino 最小系统板。

主要内容：

  • 焊接工具：烙铁、焊锡丝、助焊剂、吸锡器
  • 焊接技巧：温度控制、焊点质量判断
  • 常见焊接问题：虚焊、桥接、冷焊
周三 · 焊接练习 先在洞洞板上练习焊接直插元件（电阻、LED）。第一个焊点很丑，焊锡堆成球。助教说要先加热焊盘再送焊锡丝。
周四 · 贴片焊接 挑战贴片焊接：0805 电阻和 SOT-23 三极管。使用镊子+烙铁，非常考验手稳。
🔥 踩坑：0805 电阻焊反了方向（虽然电阻没有极性，但养成好习惯很重要）。虚焊导致电路不通，用万用表逐点检测才找到。
周五 · 完成最小系统板 焊接 ATmega328P 最小系统板：芯片、晶振、电容、USB 接口。上电后 LED 亮了！
✅ 成果：手工焊接的 Arduino 最小系统板可以正常工作

💻 Week 6 · 嵌入式编程

本周目标： 学习 Arduino/ESP32 编程，掌握 GPIO 控制与串口通信。

主要内容：

  • Arduino IDE 安装与配置
  • GPIO 输入输出：digitalRead/Write、analogRead/Write
  • 串口通信：Serial.begin()、波特率配置
  • ESP32 特性：WiFi、蓝牙、多核
周三 · 点亮第一个 LED 写了人生第一个嵌入式程序：Blink。GPIO 输出高低电平，LED 闪烁。虽然简单，但看到硬件被代码控制的感觉很神奇。
周四 · 串口通信 学习 Serial.println()，在串口监视器里看到 "Hello World"。理解了波特率的概念。
🔥 踩坑：串口监视器乱码，波特率设置错误。代码写的 115200，监视器默认 9600。
周五 · PWM 控制 学习 analogWrite() 实现 PWM 输出，控制 LED 呼吸灯效果。理解了占空比的概念。
✅ 成果：掌握了嵌入式编程基础，可以控制 GPIO 和串口

📡 Week 7 · 输入设备（传感器）

本周目标： 学习常见传感器的原理与数据采集方法。

主要内容：

  • 超声波传感器 HC-SR04：距离测量原理
  • 温湿度传感器 DHT11：单总线协议
  • 光敏电阻：模拟量采集
  • 红外避障传感器：数字量输入
周三 · HC-SR04 超声波 学习超声波测距原理：发送 10us 触发信号，测量回波时间，计算距离（距离 = 时间 × 声速 / 2）。第一次测到 25.3cm，非常准确。
周四 · DHT11 温湿度 连接 DHT11，读取温湿度数据。单总线协议时序很严格，延时不对就读不到数据。
🔥 踩坑：DHT11 数据线没有上拉电阻，读数一直失败。加了 4.7kΩ 上拉后正常。
周五 · 传感器综合测试 同时连接超声波、温湿度、光敏电阻，串口打印所有数据。学会了 millis() 非阻塞延时。
✅ 成果：掌握了多种传感器的数据采集方法

💡 Week 8 · 输出设备（执行器）

本周目标： 学习常见输出设备的驱动方法。

主要内容：

  • LED 驱动：限流电阻计算、RGB 调色
  • 蜂鸣器：有源 vs 无源、PWM 音调
  • OLED 显示屏：I2C 通信、SSD1306 驱动
  • 舵机控制：PWM 角度控制
周三 · LED 与蜂鸣器 RGB LED 混色实验：红+绿=黄，红+蓝=紫。用 tone() 函数驱动蜂鸣器播放简单旋律。
周四 · OLED 显示屏 连接 SSD1306 OLED，显示文字和图形。I2C 地址扫描确认是 0x3C。
🔥 踩坑：OLED 花屏，I2C 地址写错（0x3D vs 0x3C）。用 I2C Scanner 程序扫描后确认正确地址。
周五 · 舵机控制 使用 Servo 库控制 SG90 舵机旋转到指定角度（0-180°）。PWM 周期 20ms，脉宽 0.5ms-2.5ms 对应 0°-180°。
✅ 成果：掌握了 LED、蜂鸣器、OLED、舵机的驱动方法
